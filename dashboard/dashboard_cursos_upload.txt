function toggleDrop(key){

  const btn=document.getElementById('fbtn-'+key);
  const isOpen=drop.classList.contains('open');
  document.querySelectorAll('.fdrop').forEach(d=>d.classList.remove('open'));

function fkey(g){ return g==='estado'?'estado':g==='mes'?'mes':'curso'; }

function syncTodos(g){
  const all=document.querySelector(`input[data-group="${g}"][data-all]`);
  if(!all) return;
  const items=[...document.querySelectorAll(`input[data-group="${g}"]:not([data-all])`)];
  const checkedCount=items.filter(s=>s.checked).length;
  all.checked=checkedCount===items.length;
  all.indeterminate=checkedCount>0&&checkedCount<items.length;
}

document.querySelectorAll('input[data-group]').forEach(cb=>{
  cb.addEventListener('change',()=>{
    const g=cb.dataset.group;
    const k=fkey(g);
    if(cb.dataset.all){
      const items=[...document.querySelectorAll(`input[data-group="${g}"]:not([data-all])`)];
      items.forEach(s=>{ s.checked=cb.checked; cb.checked?F[k].add(s.value):F[k].delete(s.value); });
    } else {
      cb.checked?F[k].add(cb.value):F[k].delete(cb.value);
      syncTodos(g);
    }
    updateBadge(g,k);
    updateAll();
  });
});

function updateBadge(g,k){
  const badge=document.getElementById('badge-'+g);
  const btn=document.getElementById('fbtn-'+g);
  const n=F[k].size;
  badge.textContent=n;
  badge.style.display=n?'inline-block':'none';
  btn.classList.toggle('has-sel',n>0);
}

function clearAll(){
  F.estado.clear();F.mes.clear();F.curso.clear();
  document.querySelectorAll('input[data-group]').forEach(cb=>{ cb.checked=false; cb.indeterminate=false; });
  ['estado','mes','curso'].forEach(g=>{
    document.getElementById('badge-'+g).style.display='none';
    document.getElementById('fbtn-'+g).classList.remove('has-sel');
  });
  updateAll();
}

function normTxt(t){return t.normalize('NFD').replace(/[̀-ͯ]/g,'').toLowerCase();}

function countKeywords(texts, keys){
  const cnt={};
  texts.forEach(t=>{
    const tn=normTxt(t);
    keys.forEach(k=>{ if(tn.includes(normTxt(k))) cnt[k]=(cnt[k]||0)+1; });
  });
  return Object.entries(cnt).sort((a,b)=>b[1]-a[1]);
}

function renderFuTags(elId, texts, color, keys){
  const el=document.getElementById(elId);
  el.innerHTML='';
  const pairs=countKeywords(texts,keys);
  if(!pairs.length){
    // show full texts as small tags
    texts.slice(0,12).forEach(t=>{
      const span=document.createElement('span');
      span.className='fu-tag';
      span.style.cssText=`background:${color}18;color:${color};`;
      span.textContent=t.length>50?t.substring(0,48)+'…':t;
      el.appendChild(span);
    });
    return;
  }
  pairs.forEach(([kw,cnt])=>{
    const span=document.createElement('span');
    span.className='fu-tag';
    span.style.cssText=`background:${color}18;color:${color};`;
    const badge=document.createElement('span');
    badge.className='fu-tag-count';
    badge.style.cssText=`background:${color}35;`;
    badge.textContent=cnt;
    span.textContent=kw.charAt(0).toUpperCase()+kw.slice(1)+' ';
    span.appendChild(badge);
    el.appendChild(span);
  });
}

function renderFuQuote(txt, cor, meta){
  const d=document.createElement('div');
  d.className='fu-quote';
  d.style.cssText=`background:${cor}10;border-left-color:${cor};color:var(--text);`;
  d.innerHTML=`"${txt}"${meta?`<div class="fu-quote-meta">${meta}</div>`:''}`;
  return d;
}

function updateFollowup(){

  const retSim=nao.filter(r=>normTxt(r.ret||'').startsWith('s'));

  document.getElementById('fu-nao-resp').textContent=nao.length||'—';
  document.getElementById('fu-retorno').textContent=nao.length?Math.round(retSim.length/nao.length*100)+'%':'—';
  document.getElementById('fu-comp-resp').textContent=foi.length||'—';
  document.getElementById('fu-aplicou-pct').textContent=foi.length?aplicList.length+'/'+foi.length:'—';

  // Motivos
  const motivosEl=document.getElementById('fu-motivos');
  motivosEl.innerHTML='';
  const motivos=nao.filter(r=>r.motivo);
  document.getElementById('fu-motivos-empty').style.display=motivos.length?'none':'block';
  motivos.forEach(r=>motivosEl.appendChild(renderFuQuote(r.motivo,'#e73a57',r.e)));

  // Interesse em retornar — barra visual
  const retVis=document.getElementById('fu-retorno-vis');
  if(nao.length){
    const pct=Math.round(retSim.length/nao.length*100);
    retVis.innerHTML=`<div style="display:flex;align-items:center;gap:10px;margin-bottom:8px"><div style="flex:1;background:var(--border);border-radius:4px;height:12px;overflow:hidden"><div style="width:${pct}%;height:100%;background:#6af7c8;border-radius:4px"></div></div><span style="font-size:.85rem;font-weight:700;color:#6af7c8">${pct}%</span></div><span style="font-size:.72rem;color:var(--muted)">${retSim.length} de ${nao.length} respondentes querem participar novamente</span>`;
  }
  // Frases de interesse
  const retTagsEl=document.getElementById('fu-retorno-tags');
  retTagsEl.innerHTML='';
  document.getElementById('fu-retorno-empty').style.display=nao.filter(r=>r.ret).length?'none':'block';
  nao.filter(r=>r.ret).forEach(r=>{
    const span=document.createElement('span');
    span.className='fu-tag';
    const retNorm=normTxt(r.ret||'');
    const col=retNorm.startsWith('s')?'#6af7c8':'#e73a57';
    span.style.cssText=`background:${col}18;color:${col};`;
    span.textContent=r.ret.length>40?r.ret.substring(0,38)+'…':r.ret;
    retTagsEl.appendChild(span);
  });

  // Dores do salão
  const allDores=[...nao.map(r=>r.dor),...foi.map(r=>r.dor)].filter(Boolean);
  document.getElementById('fu-dores-empty').style.display=allDores.length?'none':'block';
  renderFuTags('fu-dores', allDores, '#7c6af7', KEYWORDS_DOR);
  const doresQuotes=document.getElementById('fu-dores-quotes');
  doresQuotes.innerHTML='';
  allDores.filter(t=>t.length>30).slice(0,4).forEach(t=>doresQuotes.appendChild(renderFuQuote(t,'#7c6af7','')));

  // Interesse em evoluir
  const allEvol=[...nao.map(r=>r.evoluir),...foi.map(r=>r.evoluir)].filter(Boolean);
  document.getElementById('fu-evoluir-empty').style.display=allEvol.length?'none':'block';
  renderFuTags('fu-evoluir', allEvol, '#f7c26a', KEYWORDS_EVOL);
  const evolQuotes=document.getElementById('fu-evoluir-quotes');
  evolQuotes.innerHTML='';
  allEvol.filter(t=>t.length>30).slice(0,4).forEach(t=>evolQuotes.appendChild(renderFuQuote(t,'#f7c26a','')));

  // Conseguiu aplicar
  const aplicEl=document.getElementById('fu-aplicou-list');
  aplicEl.innerHTML='';
  document.getElementById('fu-aplicou-empty').style.display=foi.filter(r=>r.aplicou).length?'none':'block';
  foi.filter(r=>r.aplicou).forEach(r=>aplicEl.appendChild(renderFuQuote(r.aplicou,'#6af7c8',r.e)));
}
